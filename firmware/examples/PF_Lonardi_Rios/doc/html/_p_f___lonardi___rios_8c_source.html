117static void ProcesarTask(void *pvParameter){
118 uint16_t global_contador = 0;
119
120 while(true){
121 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
122 switch(estado){
123 case esperando:
124 LedOff(LED_2);
125 LedOff(LED_3);
126 LedOn(LED_1);
127
128 if(senial_medida > umbral ){
129 estado = tiempo;
130 }
131
132 break;
133
134 case tiempo:
135 LedOn(LED_2);
136 LedOff(LED_1);
137 LedOff(LED_3);
138 vTaskDelay(UMBRAL_TEMPORAL/ portTICK_PERIOD_MS);
139 if (senial_medida > umbral){
140 estado = cebar;
141 }
142 else{
143
144 estado = esperando;
145 }
146 break;
147
148 case cebar:
149 LedOn(LED_3);
150 LedOff(LED_2);
151 LedOff(LED_1);
152
153 //Verifico que se haya configurado de forma externa el tiempo de cebado
154 if (config_tiempo_cebado > 0 ){
155 //Alarma de apertura
156 BuzzerPlayTone(1840, 150);
157 BuzzerPlayTone(1940, 150);
158 BuzzerPlayTone(2040, 150);
159 GPIOOn(GPIO_1);
160 }
161 global_contador++;
162
163 if (global_contador >= 2) {
164 estado = alerta;
165 } else {
166 if (config_tiempo_cebado > 0 ){
167 vTaskDelay((config_tiempo_cebado*1000) / portTICK_PERIOD_MS);
168 //apertura = false;
169 GPIOOff(GPIO_1);
170
171 //alarma de cierre
172 BuzzerPlayTone(2040, 150);
173 BuzzerPlayTone(1940, 150);
174 BuzzerPlayTone(1840, 150);
175
176 estado = esperando;
177 }
178 else {
179 vTaskDelay(TIEMPO_CEBADO / portTICK_PERIOD_MS);
180 //apertura = false;
181 estado = esperando;
182 }
183 }
184 break;
185 case alerta:
186 LedOn(LED_1);
187 LedOn(LED_2);
188 LedOn(LED_3);
189 // Cierra la válvula como medida de seguridad
190 GPIOOff(GPIO_1);
191
192 //alarma de alerta
193 BuzzerPlayTone(1440, 150);
194 BuzzerPlayTone(1340, 150);
195 BuzzerPlayTone(1240, 150);
196
197
198
199 vTaskDelay(1000 / portTICK_PERIOD_MS);
200 if(senial_medida < umbral ){
201 estado = esperando;
202 global_contador = 0;
203 }
204 break;
205 }
206 }
